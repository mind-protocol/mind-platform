def to_rgb(im):
    if im.mode != "RGB":
        return im.convert("RGB")
    return im

def screen(a, b):
    # a,b float 0..1
    return 1 - (1-a)*(1-b)

def overlay(a, b):
    # overlay: if a<0.5 -> 2ab else 1-2(1-a)(1-b), using a as base
    out = np.empty_like(a)
    mask = a <= 0.5
    out[mask] = 2*a[mask]*b[mask]
    out[~mask] = 1 - 2*(1-a[~mask])*(1-b[~mask])
    return out

def soft_light(a, b):
    # Photoshop-like soft light with b as blend, a as base
    out = np.empty_like(a)
    mask = b <= 0.5
    out[mask] = a[mask] - (1 - 2*b[mask]) * a[mask] * (1 - a[mask])
    out[~mask] = a[~mask] + (2*b[~mask] - 1) * (np.sqrt(a[~mask]) - a[~mask])
    return np.clip(out, 0, 1)

def blend(base_img, top_img, mode="screen", opacity=1.0):
    base = np.asarray(to_rgb(base_img)).astype(np.float32)/255.0
    top  = np.asarray(to_rgb(top_img)).astype(np.float32)/255.0
    if mode == "screen":
        mixed = screen(base, top)
    elif mode == "lighten":
        mixed = np.maximum(base, top)
    elif mode == "overlay":
        mixed = overlay(base, top)
    elif mode == "soft_light":
        mixed = soft_light(base, top)
    else:
        mixed = top
    out = base*(1-opacity) + mixed*opacity
    out = np.clip(out, 0, 1)
    return Image.fromarray((out*255).astype(np.uint8), mode="RGB")

def radial_mask(size, center=(0.58,0.44), radius=0.90, softness=0.55, invert=False):
    """
    Returns L-mode mask (0..255).
    radius: fraction of max dimension defining where it fades to ~0
    softness: 0..1, higher = softer falloff
    """
    w,h = size
    cx,cy = center[0]*w, center[1]*h
    maxr = radius*max(w,h)*0.5*2  # radius in pixels (approx)
    y, x = np.ogrid[:h, :w]
    dist = np.sqrt((x-cx)**2 + (y-cy)**2)
    t = dist / maxr
    # Smoothstep-ish falloff
    s = np.clip((t - (1-softness)) / max(softness, 1e-6), 0, 1)
    mask = 1 - s  # 1 at center, 0 at edge
    if invert:
        mask = 1 - mask
    return Image.fromarray((mask*255).astype(np.uint8), mode="L")

def luminance(rgb_):
    return (0.2126*rgb_[...,0] + 0.7152*rgb_[...,1] + 0.0722*rgb_[...,2]).astype(np.float32)

lum = luminance(rgb)

# 1) Highlight compression (tame the core)
hmask = np.clip((lum - 0.70) / (1.0 - 0.70), 0, 1)
hmask = cv2.GaussianBlur(hmask, (0,0), 12)

def compress_highlights(x, strength=0.65):
    compressed = np.log1p(8*x) / np.log1p(8)   # log-ish rolloff
    return (1-strength)*x + strength*compressed

compressed = compress_highlights(rgb, strength=0.65)
rgb_hc = np.clip(rgb*(1-hmask[...,None]) + compressed*hmask[...,None], 0, 1)

# 2) Midtone micro-contrast (Milky Way texture)
lum2 = luminance(rgb_hc)
mmask = np.exp(-((lum2-0.45)**2) / (2*(0.18**2))).astype(np.float32)
